/**
 * Projection d'un domaine de données vers une plage de pixels, et son
 * inverse. L'inverse sert au survol : retrouver la donnée sous le curseur.
 */
export interface Scale<T> {
  map(value: T): number
  invert(pixel: number): T
  readonly domain: [T, T]
  readonly range: [number, number]
}

export function linearScale(domain: [number, number], range: [number, number]): Scale<number> {
  const [d0, d1] = domain
  const [r0, r1] = range
  const span = d1 - d0
  // Domaine plat : toute valeur se projette au début de la plage. Renvoyer
  // NaN par division par zéro ferait disparaître silencieusement le tracé.
  const ratio = span === 0 ? 0 : (r1 - r0) / span
  return {
    domain,
    range,
    map: (value) => r0 + (value - d0) * ratio,
    invert: (pixel) => (ratio === 0 ? d0 : d0 + (pixel - r0) / ratio),
  }
}

export function timeScale(domain: [Date, Date], range: [number, number]): Scale<Date> {
  const numeric = linearScale([domain[0].getTime(), domain[1].getTime()], range)
  return {
    domain,
    range,
    map: (value) => numeric.map(value.getTime()),
    invert: (pixel) => new Date(numeric.invert(pixel)),
  }
}
```

export interface BandScale {
  map(value: string): number
  invert(pixel: number): string | null
  readonly bandwidth: number
  readonly domain: string[]
  readonly range: [number, number]
}

/**
 * Répartit des catégories sur une plage de pixels. `padding` est la part de
 * chaque intervalle laissée en gouttière, entre 0 et 1.
 */
export function bandScale(domain: string[], range: [number, number], padding = 0): BandScale {
  const [r0, r1] = range
  const step = domain.length === 0 ? 0 : (r1 - r0) / domain.length
  // Hors de [0, 1], le rembourrage produirait une largeur de bande négative,
  // donc un attribut SVG invalide.
  const gutter = Math.min(Math.max(padding, 0), 1)
  const bandwidth = step * (1 - gutter)
  return {
    domain,
    range,
    bandwidth,
    map: (value) => {
      const index = domain.indexOf(value)
      // Catégorie absente : rendre `r0` la poserait exactement sur la première
      // bande, et la barre fantôme se superposerait à une barre légitime sans
      // aucun signal. `NaN` rend l'attribut SVG invalide, donc rien ne se trace
      // — l'erreur se voit au lieu de se cacher. `invert` signale déjà par
      // `null` dans l'autre sens.
      return index === -1 ? Number.NaN : r0 + index * step
    },
    invert: (pixel) => {
      if (step === 0) return null
      const index = Math.floor((pixel - r0) / step)
      return index >= 0 && index < domain.length ? domain[index] : null
    },
  }
}
```

### Task 3: `niceTicks` — graduations lisibles

**Files:**

- Create: `projects/gabarit/src/lib/primitives/ticks.ts`
- Test: `projects/gabarit/src/lib/primitives/ticks.spec.ts`

**Interfaces:**

- Consumes: rien.
- Produces: `function niceTicks(min: number, max: number, count: number): number[]`

Premier des deux points durs. L'algorithme tient en vingt lignes et c'est exactement le genre de code qu'on écrit mal à la va-vite : pas brut arrondi à une puissance de dix multipliée par 1, 2, 5 ou 10, puis extension du domaine aux bornes rondes.

**Deux pièges à traiter explicitement**, parce qu'ils ne se voient qu'à l'exécution :

- **La virgule flottante.** Accumuler `v += 0.2` produit `0.6000000000000001`. Chaque graduation doit être arrondie à la précision du pas.
- **Le domaine plat.** `min === max` fait diviser par zéro. Convention retenue : une seule graduation, à la valeur. Un axe à une graduation est honnête ; un axe vide ou `NaN` ne l'est pas.

describe('niceTicks', () => {
  it.each([
    [0, 97, 5, [0, 20, 40, 60, 80, 100]],
    [0, 1, 5, [0, 0.2, 0.4, 0.6, 0.8, 1]],

    expect(niceTicks(min, max, count)).toEqual(expected)
  })

  it("d'un domaine plat, rend une seule graduation", () => {
    expect(niceTicks(5, 5, 5)).toEqual([5])
  })

  it('ne produit jamais de flottant accumulé', () => {
    // 0.6000000000000001 est le résultat naïf de 0.2 + 0.2 + 0.2.
    const ticks = niceTicks(0, 1, 5)
    expect(ticks).toEqual([0, 0.2, 0.4, 0.6, 0.8, 1])
    for (const tick of ticks) {
      expect(String(tick).length).toBeLessThanOrEqual(3)
    }
  })

  it('englobe toujours le domaine demandé', () => {
    for (const [min, max] of [
      [3, 97],
      [-13, 41],
      [0.03, 0.71],
    ] as const) {
      const ticks = niceTicks(min, max, 5)
      expect(ticks[0]).toBeLessThanOrEqual(min)
      expect(ticks[ticks.length - 1]).toBeGreaterThanOrEqual(max)
    }
  })

  it('rend des graduations régulièrement espacées', () => {
    const ticks = niceTicks(0, 97, 5)
    const gaps = ticks.slice(1).map((t, i) => t - ticks[i])
    expect(new Set(gaps).size).toBe(1)
  })

  it('remet des bornes inversées dans l’ordre plutôt que de rendre un axe vide', () => {
    expect(niceTicks(10, 5, 5)).toEqual(niceTicks(5, 10, 5))
    expect(niceTicks(10, 5, 5).length).toBeGreaterThan(0)
  })
})

```typescript
/**
 * Graduations lisibles couvrant `[min, max]`, en visant `count` intervalles.
 * On ne veut pas `0 / 3,7 / 7,4` mais `0 / 5 / 10` : le pas brut est donc
 * arrondi à une puissance de dix multipliée par 1, 2, 5 ou 10, puis le
 * domaine est étendu aux bornes rondes les plus proches.
 *
 * Le résultat englobe toujours le domaine demandé — un axe qui coupe la
 * donnée qu'il est censé porter serait pire qu'un axe laid.
 */
export function niceTicks(min: number, max: number, count: number): number[] {
  if (!Number.isFinite(min) || !Number.isFinite(max)) return []
  // Domaine plat : une seule graduation. Diviser par zéro rendrait NaN, et
  // un axe vide masquerait la donnée au lieu de la situer.
  if (min === max) return [min]
  // Bornes inversées : un pas négatif rendrait NaN puis un tableau vide, et
  // l'axe disparaîtrait en silence. On les remet dans l'ordre — les graduations
  // sont croissantes quel que soit le sens dans lequel on décrit le domaine.
  if (max < min) return niceTicks(max, min, count)

  const rawStep = (max - min) / Math.max(1, count)
  const magnitude = 10 ** Math.floor(Math.log10(rawStep))
  const normalized = rawStep / magnitude
  const step = magnitude * (normalized <= 1 ? 1 : normalized <= 2 ? 2 : normalized <= 5 ? 5 : 10)

  const first = Math.floor(min / step) * step
  const last = Math.ceil(max / step) * step
  // La précision du pas borne celle des graduations : sans cet arrondi,
  // l'accumulation produit 0.6000000000000001.
  const decimals = Math.max(0, -Math.floor(Math.log10(step)))

  const ticks: number[] = []
  // La tolérance absorbe l'erreur d'accumulation sur la borne haute.
  for (let value = first; value <= last + step / 1e9; value += step) {
    ticks.push(Number(value.toFixed(decimals)))
  }
  return ticks
}
```

### Task 4: `timeTicks` — graduations temporelles

**Files:**

- Modify: `projects/gabarit/src/lib/primitives/ticks.ts`
- Modify: `projects/gabarit/src/lib/primitives/ticks.spec.ts`

**Interfaces:**

- Consumes: rien.
- Produces:
  - `type TimeUnit = 'second' | 'minute' | 'hour' | 'day' | 'month' | 'year'`
  - `interface TimeTicks { values: Date[]; unit: TimeUnit; format(date: Date, locale: string): string }`
  - `function timeTicks(start: Date, end: Date, pixelWidth: number): TimeTicks`

Second point dur. Un axe qui affiche « 03/09 03/09 03/09 » parce que la granularité est trop fine pour la largeur disponible est le bug visuel classique de tout graphique fait maison. La fonction ne rend donc pas seulement des instants : elle rend **la granularité choisie et le format qui va avec**, pour que l'appelant ne puisse pas les désaccorder.

**Échelle de pas**, du plus fin au plus grossier, en millisecondes : 1 s, 5 s, 15 s, 30 s, 1 min, 5 min, 15 min, 30 min, 1 h, 3 h, 6 h, 12 h, 1 j, 2 j, 7 j, 1 mois (30 j), 3 mois, 1 an (365 j). On retient le premier pas dont le nombre de graduations tient dans `pixelWidth / 60` — soixante pixels étant la largeur minimale sous laquelle deux étiquettes de date se chevauchent.

describe('timeTicks', () => {

  it('choisit les mois sur deux ans', () => {

    expect(result.unit).toBe('month')

  it("sur une largeur étroite, s'élargit plutôt que d'entasser", () => {
    const large = timeTicks(at('2026-01-01T00:00:00Z'), at('2026-01-02T00:00:00Z'), 800)

    for (const width of [120, 300, 600, 1200]) {

          const result = timeTicks(start, end, width)
          const labels = result.values.map((v) => result.format(v, 'fr-FR'))
          for (let i = 1; i < labels.length; i++) {
            expect(
              labels[i],
              `${days} j, ${width} px, décalage ${offset}, unité ${result.unit}`,
            ).not.toBe(labels[i - 1])
          }
        }
      }
    }
  })

  it('respecte le budget sur toutes les granularités, repli compris', () => {
    for (const days of [0.01, 0.5, 3, 40, 150, 900, 5000, 70000]) {
      for (const width of [120, 200, 400, 600, 900, 1200]) {

          result.values.length,
          `${days} j, ${width} px, unité ${result.unit}`,

    const result = timeTicks(at('2021-04-03T00:00:00Z'), at('2021-08-31T00:00:00Z'), 600)

      expect(value.getUTCDate()).toBe(1)
      expect(value.getUTCHours()).toBe(0)

  })

    const meme = at('2026-01-01T00:00:00Z')

- [ ] **Step 3: Écrire l'implémentation**

Le format s'appuie sur `Intl.DateTimeFormat`, avec la locale **toujours passée en paramètre** : une locale déduite de l'environnement rendrait les tests dépendants de la machine et la librairie implicitement localisée.

```typescript
export type TimeUnit = 'second' | 'minute' | 'hour' | 'day' | 'month' | 'year'

export interface TimeTicks {
  values: Date[]
  unit: TimeUnit
  format(date: Date, locale: string): string
}

const SECOND = 1000
const MINUTE = 60 * SECOND
const HOUR = 60 * MINUTE
const DAY = 24 * HOUR

/**
 * Du plus fin au plus grossier. `unit` pilote le format des étiquettes.
 *
 * `months` marque les pas qui doivent être parcourus en calendrier et non en
 * millisecondes : un pas de « 30 jours » ne tombe jamais sur un début de mois,
 * si bien que deux graduations peuvent partager un même mois de 31 jours et
 * que l'axe affiche deux fois « août 2021 ». `ms` reste la durée approchée du
 * pas, qui suffit à choisir la granularité.
 */
const STEPS: { ms: number; unit: TimeUnit; months?: number }[] = [
  { ms: SECOND, unit: 'second' },
  { ms: 5 * SECOND, unit: 'second' },
  { ms: 15 * SECOND, unit: 'second' },
  { ms: 30 * SECOND, unit: 'second' },
  { ms: MINUTE, unit: 'minute' },
  { ms: 5 * MINUTE, unit: 'minute' },
  { ms: 15 * MINUTE, unit: 'minute' },
  { ms: 30 * MINUTE, unit: 'minute' },
  { ms: HOUR, unit: 'hour' },
  { ms: 3 * HOUR, unit: 'hour' },
  { ms: 6 * HOUR, unit: 'hour' },
  { ms: 12 * HOUR, unit: 'hour' },
  { ms: DAY, unit: 'day' },
  { ms: 2 * DAY, unit: 'day' },
  { ms: 7 * DAY, unit: 'day' },
  { ms: 30 * DAY, unit: 'month', months: 1 },
  { ms: 90 * DAY, unit: 'month', months: 3 },
  { ms: 365 * DAY, unit: 'year', months: 12 },
]

const FORMATS: Record<TimeUnit, Intl.DateTimeFormatOptions> = {
  second: { hour: '2-digit', minute: '2-digit', second: '2-digit' },
  minute: { hour: '2-digit', minute: '2-digit' },
  hour: { hour: '2-digit', minute: '2-digit' },
  day: { day: '2-digit', month: '2-digit' },
  month: { month: 'short', year: 'numeric' },
  year: { year: 'numeric' },
}

/** Largeur minimale d'une étiquette de date avant chevauchement. */
const MIN_PIXELS_PER_TICK = 60

export function timeTicks(start: Date, end: Date, pixelWidth: number): TimeTicks {
  const span = end.getTime() - start.getTime()
  const makeFormatter = (unit: TimeUnit) => (date: Date, locale: string) =>
    new Intl.DateTimeFormat(locale, { ...FORMATS[unit], timeZone: 'UTC' }).format(date)

  if (span <= 0) {
    return { values: [start], unit: 'day', format: makeFormatter('day') }
  }

  const maxTicks = Math.max(2, Math.floor(pixelWidth / MIN_PIXELS_PER_TICK))
  // Premier pas assez large pour que le compte tienne. Les bornes peuvent
  // toutes deux tomber sur une graduation, auquel cas la suite en compte
  // span/pas + 1.
  //
  // Quand aucun pas de la table ne tient, on éclaircit le pas annuel dans la
  // proportion qui manque plutôt que de le retenir tel quel : sans ça, trente
  // ans à 120 px rendaient trente graduations pour un budget de deux.
  const coarsest = STEPS[STEPS.length - 1]
  const chosen = STEPS.find((step) => span / step.ms + 1 <= maxTicks) ?? {
    ...coarsest,
    months: coarsest.months! * Math.max(1, Math.ceil((span / coarsest.ms + 1) / maxTicks)),
  }

  const values: Date[] = []
  if (chosen.months) {
    // Marche calendaire : chaque graduation tombe sur un premier du mois, donc
    // deux graduations ne peuvent plus partager la même étiquette mensuelle.
    const step = chosen.months
    const cursor =
      step >= 12
        ? new Date(Date.UTC(Math.floor(start.getUTCFullYear() / (step / 12)) * (step / 12), 0, 1))
        : new Date(
            Date.UTC(start.getUTCFullYear(), Math.floor(start.getUTCMonth() / step) * step, 1),
          )
    while (cursor.getTime() < start.getTime()) cursor.setUTCMonth(cursor.getUTCMonth() + step)
    while (cursor.getTime() <= end.getTime()) {
      values.push(new Date(cursor))
      cursor.setUTCMonth(cursor.getUTCMonth() + step)
    }
  } else {
    const first = Math.ceil(start.getTime() / chosen.ms) * chosen.ms
    for (let t = first; t <= end.getTime(); t += chosen.ms) {
      values.push(new Date(t))
    }
  }
  // Un intervalle plus court que le pas ne produirait aucune graduation :
  // on borne alors sur le début, pour que l'axe situe quand même la donnée.
  if (values.length === 0) values.push(start)

  return { values, unit: chosen.unit, format: makeFormatter(chosen.unit) }
}
```

```typescript
export function formatNumber(value: number, locale: string, decimals?: number): string {
  return new Intl.NumberFormat(locale, {
    minimumFractionDigits: decimals,
    maximumFractionDigits: decimals,
  }).format(value)
}

/** 1200 → « 1,2 k ». Pour les axes et les tuiles, où la place manque. */
export function formatCompact(value: number, locale: string): string {
  return new Intl.NumberFormat(locale, {
    notation: 'compact',
    maximumFractionDigits: 1,
  }).format(value)
}

/**
 * Durée lisible : deux unités au plus, la plus grande d'abord. Au-delà de
 * deux, l'œil ne lit plus une durée mais une suite de nombres.
 *
 * La seconde unité est celle qui suit positionnellement la première, et elle
 * est omise quand elle vaut zéro : une heure et trente secondes rend « 1 h ».
 * Sauter un rang pour afficher « 1 h 30 s » se lirait en français comme
 * 1 h 30 — une demi-heure d'erreur pour épargner 0,8 % d'arrondi.
 *
 * Les symboles d'unité (`ms`, `s`, `min`, `h`) sont ceux du SI et ne se
 * traduisent pas ; les nombres, eux, passent tous par la locale reçue.
 */
export function formatDuration(ms: number, locale: string): string {
  if (ms < 1000) return `${formatNumber(Math.round(ms), locale)} ms`

  const totalSeconds = Math.round(ms / 1000)
  const hours = Math.floor(totalSeconds / 3600)
  const minutes = Math.floor((totalSeconds % 3600) / 60)
  const seconds = totalSeconds % 60
  const num = (value: number) => formatNumber(value, locale)

  if (hours > 0) return minutes > 0 ? `${num(hours)} h ${num(minutes)} min` : `${num(hours)} h`
  if (minutes > 0)
    return seconds > 0 ? `${num(minutes)} min ${num(seconds)} s` : `${num(minutes)} min`
  return `${num(seconds)} s`
}

/** Reçoit un ratio (0,1234), pas un pourcentage déjà multiplié. */
export function formatPercent(ratio: number, locale: string, decimals = 0): string {
  return new Intl.NumberFormat(locale, {
    style: 'percent',
    minimumFractionDigits: decimals,
    maximumFractionDigits: decimals,
  }).format(ratio)
}
```

```typescript
export interface Point {
  x: number
  y: number
}

/** Deux décimales suffisent au pixel près, et évitent des `d` interminables. */
const round = (n: number): number => Number(n.toFixed(2))

export function linePath(points: Point[]): string {
  if (points.length === 0) return ''
  return points
    .map((point, index) => `${index === 0 ? 'M' : 'L'}${round(point.x)},${round(point.y)}`)
    .join('')
}

/** Même tracé que `linePath`, refermé sur `baseline` pour être rempli. */
export function areaPath(points: Point[], baseline: number): string {
  if (points.length === 0) return ''
  const last = points[points.length - 1]
  const first = points[0]
  return `${linePath(points)}L${round(last.x)},${round(baseline)}L${round(first.x)},${round(baseline)}Z`
}
```
